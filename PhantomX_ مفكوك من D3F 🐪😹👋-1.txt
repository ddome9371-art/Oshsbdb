---Create an instance, apply properties, then parent it.
---@param className string
---@param properties table
---@param parent Instance?
---@return Instance
local function create(className, properties, parent)
	local object = Instance.new(className)

	for property, value in pairs(properties) do
		object[property] = value
	end

	object.Parent = parent
	return object
end

---Send a request through the executor request function.
---@param options table
---@return table?
local function httpRequest(options)
	local requestFunction = request or http_request or (syn and syn.request) or (http and http.request)

	if not requestFunction then
		return nil
	end

	local success, response = pcall(requestFunction, options)
	return success and response or nil
end

---Turn a named signal connection on or off.
---@param name string
---@param enabled boolean
---@param signal RBXScriptSignal?
---@param callback function?
local function setLoop(name, enabled, signal, callback)
	if loops[name] then
		loops[name]:Disconnect()
		loops[name] = nil
	end

	if enabled then
		loops[name] = signal:Connect(callback)
	end
end

---Remove every leftover WindUI gui, a half destroyed one breaks new windows.
local function cleanWindUiGuis()
	for _, parent in ipairs({ getGuiParent(), coreGui, playerGui }) do
		pcall(function()
			for _, child in ipairs(parent:GetChildren()) do
				if child.Name:sub(1, 6) == "WindUI" then
					child:Destroy()
				end
			end
		end)
	end
end

---Show or hide the window, every WindUI version is handled.
local function toggleWindow()
	-- The window never loaded, build it again.
	if not window then
		return rebuildUi()
	end

	if type(window.Toggle) == "function" and pcall(window.Toggle, window) then
		return
	end

	local method = window.Closed and window.Open or window.Close

	if type(method) == "function" and pcall(method, window) then
		return
	end

	-- Last resort, flip the WindUI gui directly.
	for _, parent in ipairs({ getGuiParent(), coreGui }) do
		local gui = parent:FindFirstChild("WindUI")

		if gui then
			gui.Enabled = not gui.Enabled
			return
		end
	end

	-- The gui is gone, build a new window.
	window = nil
	rebuildUi()
end

-- License.

function License.getKey()
	local path = CONFIG.folder .. "/license.txt"
	local fileKey = isfile and isfile(path) and readfile(path) or nil
	fileKey = fileKey and fileKey:match("^%s*(.-)%s*$") or nil
	return getgenv().PhantomXLicenseKey or fileKey
end

function License.saveKey(key)
	getgenv().PhantomXLicenseKey = key

	pcall(function()
		if makefolder and not isfolder(CONFIG.folder) then
			makefolder(CONFIG.folder)
		end

		writefile(CONFIG.folder .. "/license.txt", tostring(key))
	end)
end

function License.heartbeat()
	if not sessionToken then
		return
	end

	httpRequest({
		Url = CONFIG.deviceCheck .. CONFIG.heartbeatPath, Method = "POST", Headers = { ["Content-Type"] = "application/json" },
		Body = httpService:JSONEncode({ sessionToken = sessionToken, productId = CONFIG.productId }),
	})
end

---Activate a license key and keep the session alive.
---@param key string
---@return boolean
function License.activate(key)
	key = key or License.getKey()

	if not key or key == "" then
		notify("License Error", translate("invalidKey"))
		return false
	end

	local response = httpRequest({
		Url = CONFIG.deviceCheck .. CONFIG.activatePath, Method = "POST", Headers = { ["Content-Type"] = "application/json" },
		Body = httpService:JSONEncode({
			licenseKey = key, productId = CONFIG.productId, userId = localPlayer.UserId,
			username = localPlayer.Name, platform = CONFIG.platforms,
		}),
	})

	if not response or not response.Body then
		notify("License Error", translate("activationFailed"))
		return false
	end

	local success, data = pcall(httpService.JSONDecode, httpService, response.Body)

	if not success or type(data) ~= "table" then
		notify("License Error", translate("badResponse"))
		return false
	end

	local errorKey = LICENSE_ERRORS[data.error] or LICENSE_ERRORS[data.status]

	if errorKey then
		notify("License Error", errorKey == "licenseRejected" and data.message or translate(errorKey))
		return false
	end

	if not (data.sessionToken or data.success or data.status == "Activated") then
		notify("License Error", translate("licenseRejected"))
		return false
	end

	sessionToken, State.licensed = data.sessionToken or "active", true
	License.saveKey(key)
	notify("PhantomX", translate("licenseVerified"))

	-- Heartbeat every minute while the session is alive, never run two loops.
	if not heartbeatRunning then
		heartbeatRunning = true

		task.spawn(function()
			while sessionToken do
				task.wait(60)
				License.heartbeat()
			end

			heartbeatRunning = false
		end)
	end

	return true
end

-- Kill effects.
local Effects = {}

local function forEachPart(character, callback)
	for _, part in ipairs(character:GetDescendants()) do
		if part:IsA("BasePart") then
			callback(part)
		end
	end
end

local function playSound(volume)
	local sound = create("Sound", { SoundId = CONFIG.explosionSound, Volume = volume }, workspace)
	sound:Play()
	debris:AddItem(sound, 5)
end

---Push the character with a temporary velocity.
local function launch(character, velocity, maxForce, duration)
	local root = character:FindFirstChild("HumanoidRootPart")

	if root then
		debris:AddItem(create("BodyVelocity", { Velocity = velocity, MaxForce = maxForce }, root), duration)
	end
end

function Effects.freeze(character)
	local backups = {}

	forEachPart(character, function(part)
		backups[part] = { part.Anchored, part.Color, part.Material }
		part.Anchored, part.Color, part.Material = true, Color3.fromRGB(150, 220, 255), Enum.Material.Ice
	end)

	task.delay(3, function()
		for part, backup in pairs(backups) do
			if part.Parent then
				part.Anchored, part.Color, part.Material = backup[1], backup[2], backup[3]
			end
		end
	end)
end

function Effects.lightning(character)
	local root = character:FindFirstChild("HumanoidRootPart")

	if not root then
		return
	end

	local bolt = create("Part", {
		Size = Vector3.new(0.3, 20, 0.3), Position = root.Position + Vector3.new(0, 10, 0), Anchored = true,
		CanCollide = false, Material = Enum.Material.Neon, Color = Color3.fromRGB(200, 220, 255),
	}, workspace)

	playSound(0.8)
	debris:AddItem(bolt, 0.4)
end

local function applyAntiLag()
	pcall(function()
		if not State.antiLagActive then
			-- Only restore what we changed, detach must not touch an untouched game.
			if antiLagBackup then
				settings().Rendering.QualityLevel = antiLagBackup.quality
				lighting.GlobalShadows = antiLagBackup.shadows
				antiLagBackup = nil
			end

			for _, object in ipairs(disabledEffects) do
				pcall(function()
					object.Enabled = true
				end)
			end

			return table.clear(disabledEffects)
		end

		antiLagBackup = antiLagBackup or { quality = settings().Rendering.QualityLevel, shadows = lighting.GlobalShadows }
		settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
		lighting.GlobalShadows = false

		for _, object in ipairs(workspace:GetDescendants()) do
			local isEffect = object:IsA("ParticleEmitter") or object:IsA("Trail") or object:IsA("Smoke") or object:IsA("Fire")

			if isEffect and object.Enabled then
				object.Enabled = false
				table.insert(disabledEffects, object)
			end
		end
	end)
end

local function applyLighting()
	for _, name in ipairs({ "PX_Bloom", "PX_ColorCorrection", "PX_SunRays", "PX_Atmosphere" }) do
		local effect = lighting:FindFirstChild(name)

		if effect then
			effect:Destroy()
		end
	end

	-- Restore whatever the game had before.
	if not State.lightingActive then
		for property, value in pairs(lightingBackup or {}) do
			lighting[property] = value
		end

		for property, value in pairs(atmosphereBackup and atmosphereBackup.values or {}) do
			atmosphereBackup.object[property] = value
		end

		lightingBackup, atmosphereBackup = nil, nil
		return
	end

	if not lightingBackup then
		lightingBackup = {}

		for _, property in ipairs(LIGHTING_PROPERTIES) do
			lightingBackup[property] = lighting[property]
		end
	end

	lighting.Brightness, lighting.ClockTime, lighting.FogEnd, lighting.GlobalShadows = 2.2, 14, 1e6, true
	lighting.OutdoorAmbient, lighting.Ambient = Color3.fromRGB(140, 140, 150), Color3.fromRGB(80, 80, 90)

	create("BloomEffect", { Name = "PX_Bloom", Intensity = 0.4, Size = 24, Threshold = 1.5 }, lighting)
	create("ColorCorrectionEffect", { Name = "PX_ColorCorrection", Brightness = 0.05, Contrast = 0.1, Saturation = 0.15 }, lighting)
	create("SunRaysEffect", { Name = "PX_SunRays", Intensity = 0.08, Spread = 0.5 }, lighting)

	local atmosphere = lighting:FindFirstChildOfClass("Atmosphere")

	if not atmosphere then
		local properties = { Name = "PX_Atmosphere" }

		for property, value in pairs(ATMOSPHERE) do
			properties[property] = value
		end

		return create("Atmosphere", properties, lighting)
	end

	if not atmosphereBackup then
		atmosphereBackup = { object = atmosphere, values = {} }

		for property in pairs(ATMOSPHERE) do
			atmosphereBackup.values[property] = atmosphere[property]
		end
	end

	for property, value in pairs(ATMOSPHERE) do
		atmosphere[property] = value
	end
end

local function applyStretch()
	local camera = workspace.CurrentCamera

	if State.stretchValue > 0 and camera then
		baseFov = baseFov or camera.FieldOfView
	end

	setLoop("stretch", State.stretchValue > 0 and baseFov ~= nil, runService.RenderStepped, function()
		workspace.CurrentCamera.FieldOfView = baseFov + State.stretchValue / 100 * 50
	end)

	if State.stretchValue <= 0 and baseFov and camera then
		camera.FieldOfView, baseFov = baseFov, nil
	end
end

---Hide or restore the head, its decals and head accessories.
local function applyHeadless()
	if not State.headlessEnabled then
		for object, transparency in pairs(hidden) do
			pcall(function()
				object.Transparency = transparency
			end)
		end

		return table.clear(hidden)
	end

	local character = getCharacter()
	local head = character and character:FindFirstChild("Head")

	if not head then
		return
	end

	local function hide(object)
		hidden[object] = hidden[object] or object.Transparency
		object.Transparency = 1
	end

	hide(head)

	for _, child in ipairs(head:GetChildren()) do
		if child:IsA("Decal") or child:IsA("Texture") then
			hide(child)
		end
	end

	for _, accessory in ipairs(character:GetChildren()) do
		local handle = accessory:IsA("Accessory") and accessory:FindFirstChild("Handle")

		for _, attachment in ipairs(handle and HEAD_ATTACHMENTS or {}) do
			if handle:FindFirstChild(attachment) then
				hide(handle)
				break
			end
		end
	end
end

---Find the closest living player inside the range.
---@param range number
---@return Player?
local function getNearestPlayer(range)
	local root = getRoot()

	if not root then
		return nil
	end

	local best, bestDistance = nil, range

	for _, player in ipairs(playersService:GetPlayers()) do
		local character = player.Character
		local targetRoot = character and character:FindFirstChild("HumanoidRootPart")
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")

		if player == localPlayer or not targetRoot or not humanoid or humanoid.Health <= 0 then
			continue
		end

		local distance = (targetRoot.Position - root.Position).Magnitude

		if distance < bestDistance then
			best, bestDistance = player, distance
		end
	end

	return best
end

---Play the selected kill effect when the current target dies.
---@param player Player?
local function watchTargetDeath(player)
	-- The old connection lingers so the death that dropped the target still counts.
	local oldConnection = deathConnection
	deathConnection = nil

	if oldConnection then
		task.delay(1, function()
			oldConnection:Disconnect()
		end)
	end

	local character = player and player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")

	if humanoid then
		deathConnection = humanoid.Died:Connect(function()
			playKillEffect(State.killEffect, character)
		end)
	end
end

local function assistStep()
	local target = State.assistEnabled and getNearestPlayer(State.assistRange) or nil

	if target ~= assistTarget then
		assistTarget, targetSince = target, os.clock()
		watchTargetDeath(target)
	end

	local head = target and target.Character and target.Character:FindFirstChild("Head")
	local camera = workspace.CurrentCamera

	if not head or not camera or os.clock() - targetSince < State.reactionTime then
		return
	end

	local alpha = math.clamp(State.assistPower / 100 * 0.2 * State.ahSmooth, 0, 1)
	local origin = camera.CFrame.Position
	local offset = head.Position - origin

	if offset.Magnitude < 0.01 then
		return
	end

	local direction = camera.CFrame.LookVector:Lerp(offset.Unit, alpha)

	camera.CFrame = CFrame.lookAt(origin, origin + direction)
end

local function shakeStep()
	local camera = workspace.CurrentCamera

	if State.camShake and camera then
		local time = os.clock() * State.camShakeSpeed * 12
		camera.CFrame = camera.CFrame * CFrame.Angles(math.sin(time) * 0.04, math.cos(time * 1.4) * 0.04, 0)
	end
end

local function orbitStep(delta)
	local root = getRoot()

	if (State.orbitEnabled or State.turnToggle) and root then
		root.CFrame = root.CFrame * CFrame.Angles(0, math.rad(State.orbitSpeed * delta * 60), 0)
	end
end

local function onRenderStep(delta)
	assistStep()
	shakeStep()
	orbitStep(delta)
end

local function speedStep()
	local humanoid, root = getHumanoid(), getRoot()

	if not humanoid or not root then
		return
	end

	local moving = humanoid.MoveDirection.Magnitude > 0
	local vertical = Vector3.new(0, root.AssemblyLinearVelocity.Y, 0)

	if State.rushEnabled and moving then
		root.AssemblyLinearVelocity = humanoid.MoveDirection * State.speedValue * State.ahRush + vertical
	elseif State.speedEnabled and State.speedMode == "Velocity" then
		if moving then
			root.AssemblyLinearVelocity = humanoid.MoveDirection * State.speedValue + vertical
		end
	elseif State.speedEnabled and humanoid.WalkSpeed ~= State.speedValue then
		humanoid.WalkSpeed = State.speedValue
	end
end

local function resetSpeed()
	local humanoid = getHumanoid()

	if humanoid then
		humanoid.WalkSpeed = 16
	end
end

---Slide along walls instead of stopping against them.
local function wallStep()
	local root = getRoot()

	if not root then
		return
	end

	wallParams.FilterDescendantsInstances = { getCharacter() }

	local result = workspace:Raycast(root.Position, root.CFrame.LookVector * 2, wallParams)

	if not result then
		return
	end

	local velocity = root.AssemblyLinearVelocity
	local into = velocity:Dot(result.Normal)

	if into < 0 then
		root.AssemblyLinearVelocity = velocity - result.Normal * into
	end
end

local function autoHoldStep()
	local character = getCharacter()

	for _, child in ipairs(character and character:GetChildren() or {}) do
		if child:IsA("Tool") then
			pcall(child.Activate, child)
		end
	end
end

---Touch nearby bombs so we pick them up.
local function grabBombStep()
	local root = getRoot()

	if not root or not firetouchinterest or os.clock() - lastGrab < 0.25 then
		return
	end

	lastGrab = os.clock()

	for _, object in ipairs(workspace:GetChildren()) do
		local name = object.Name:lower()
		local isBomb = name:find("bomb") or name:find("grenade") or name:find("tnt") or name:find("explosive")

		if not isBomb or not (object:IsA("BasePart") or object:IsA("Model")) then
			continue
		end

		local part = object:IsA("BasePart") and object or object:FindFirstChildWhichIsA("BasePart", true)

		if part and (part.Position - root.Position).Magnitude < 20 then
			firetouchinterest(part, root, 0)
			firetouchinterest(part, root, 1)
		end
	end
end

local function applyLoops()
	setLoop("walls", State.wallsEnabled, runService.Heartbeat, wallStep)
	setLoop("autoHold", State.autoHoldActive, runService.Heartbeat, autoHoldStep)
	setLoop("grabBomb", State.grabBombActive, runService.Heartbeat, grabBombStep)
end

-- Floating gui.

---Create a draggable round button, the click only counts if it was not dragged.
---@param name string
---@param order number
---@param properties table
---@param onClick function
local function createFloatingButton(name, order, properties, onClick)
	local gui = create("ScreenGui", { Name = name, ResetOnSpawn = false, IgnoreGuiInset = true, DisplayOrder = order, Enabled = not State.recordMode }, getGuiParent())

	properties.TextColor3, properties.Font, properties.AutoButtonColor = Color3.new(1, 1, 1), Enum.Font.GothamBold, true

	local button = create("TextButton", properties, gui)
	create("UICorner", { CornerRadius = UDim.new(1, 0) }, button)

	local moved = false

	button.InputBegan:Connect(function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		local origin, startPosition = input.Position, button.Position
		local changed, ended

		moved = false

		changed = userInputService.InputChanged:Connect(function(change)
			if change.UserInputType ~= Enum.UserInputType.MouseMovement and change.UserInputType ~= Enum.UserInputType.Touch then
				return
			end

			local delta = change.Position - origin
			moved = moved or delta.Magnitude > 8

			if moved then
				button.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
			end
		end)

		ended = input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				changed:Disconnect()
				ended:Disconnect()
			end
		end)
	end)

	-- Activated is the most reliable click on both mouse and touch.
	button.Activated:Connect(function()
		if not moved then
			onClick(button)
		end
	end)

	return gui, button
end

-- Config.

local function saveConfig(name)
	if not writefile then
		return notify("Config", translate("failed"))
	end

	local success = pcall(function()
		if makefolder and not isfolder(CONFIG.folder) then
			makefolder(CONFIG.folder)
		end

		writefile(CONFIG.folder .. "/" .. (type(name) == "string" and name or "config") .. ".json", httpService:JSONEncode(State))
	end)

	notify("Config", translate(success and "saved" or "failed"))
end

local function loadConfig(name)
	local path = CONFIG.folder .. "/" .. (type(name) == "string" and name or "config") .. ".json"

	if not (isfile and isfile(path)) then
		notify("Config", translate("failed"))
		return nil
	end

	local success, data = pcall(function()
		return httpService:JSONDecode(readfile(path))
	end)

	if not success or type(data) ~= "table" then
		return nil
	end

	for key, value in pairs(data) do
		if type(State[key]) == type(value) and key ~= "licensed" then
			State[key] = value
		end
	end

	applyAll()
	task.defer(rebuildUi)
	notify("Config", translate("loaded"))
	return data
end

---Apply the selected theme to WindUI, the call is best effort.
local function applyTheme()
	for _, theme in ipairs(THEMES) do
		if theme[1] == State.theme and windUi then
			local background, accent = Color3.fromRGB(table.unpack(theme[3])), Color3.fromRGB(table.unpack(theme[4]))

			pcall(function()
				windUi:AddTheme({
					Name = theme[1], Accent = accent, Dialog = background, Outline = accent, Background = background,
					Text = theme[1] == "Light" and Color3.new(0, 0, 0) or Color3.new(1, 1, 1),
					Placeholder = Color3.fromRGB(140, 140, 150), Button = background:Lerp(accent, 0.2), Icon = accent,
				})
				windUi:SetTheme(theme[1])
			end)
		end
	end
end

---Select a theme by id or label.
---@param idOrName string
local function selectTheme(idOrName)
	for _, theme in ipairs(THEMES) do
		if theme[1] == idOrName or theme[2] == idOrName then
			State.theme = theme[1]
			applyTheme()
			return theme
		end
	end

	return THEMES[1]
end

local function addDropdown(tab, labelKey, values, current, onChange)
	return tab:Dropdown({
		Title = translate(labelKey), Desc = describe(labelKey), Values = values, Value = current,
		Callback = function(value)
			onChange(tostring(value))
		end,
	})
end

local function addButton(tab, labelKey, callback)
	return tab:Button({ Title = translate(labelKey), Desc = describe(labelKey), Callback = callback })
end

local function addInput(tab, labelKey, field)
	return tab:Input({
		Title = translate(labelKey), Desc = describe(labelKey), Placeholder = translate(labelKey), Value = State[field],
		Callback = function(value)
			State[field] = tostring(value)
		end,
	})
end

---Build the whole WindUI window from the current state.
local function buildUi()
	window = windUi:CreateWindow({ Title = CONFIG.name, Author = "PhantomX", Folder = CONFIG.folder })
	PhantomX.Window = window

	local main = window:Tab({ Title = translate("mainFeatures") })

	main:Section({ Title = translate("assist") })
	assistToggle = addToggle(main, "enableAssist", "assistEnabled", refreshAssistButton)
	addSlider(main, "assistPower", "assistPower", 0, 100, 1)
	addDropdown(main, "assistRange", RANGE_VALUES, tostring(State.assistRange), function(value)
		setAHDist(value)
	end)

	main:Section({ Title = translate("speed") })
	addToggle(main, "enableSpeed", "speedEnabled", function(value)
		if not value then
			resetSpeed()
		end
	end)
	addSlider(main, "speedLevel", "speedValue", 16, 250, 1)
	addDropdown(main, "speedType", { "WalkSpeed", "Velocity" }, State.speedMode, function(value)
		if value == State.speedMode then
			return
		end

		State.speedMode = value
		resetSpeed()
	end)

	main:Section({ Title = translate("camera") })
	addToggle(main, "enableShake", "camShake")
	addSlider(main, "shakeSpeed", "camShakeSpeed", 1, 20, 1)
	addSlider(main, "camBoost", "camBoost", 0.1, 5, 0.1, applySensitivity)

	main:Section({ Title = translate("movement") })
	addToggle(main, "enableOrbit", "orbitEnabled", function(value)
		State.turnToggle = value
	end)
	addSlider(main, "orbitSpeed", "orbitSpeed", 1, 30, 1, setAHOrbit)
	addToggle(main, "enableRush", "rushEnabled")
	addToggle(main, "enableSmooth", "smoothEnabled", function(value)
		setAHSmooth(value and SMOOTH_BOOST or 1)
	end)

	main:Section({ Title = translate("extras") })
	addToggle(main, "enableAutoHold", "autoHoldActive", applyLoops)
	addToggle(main, "enableGrabBomb", "grabBombActive", applyLoops)
	addToggle(main, "enableWalls", "wallsEnabled", applyLoops)

	local appearance = window:Tab({ Title = translate("appearance") })

	addToggle(appearance, "enableHeadless", "headlessEnabled", applyHeadless)
	addToggle(appearance, "enableKorblox", "korbloxEnabled", function()
		task.spawn(applyKorblox)
	end)
	addInput(appearance, "username", "skinUsername")
	addButton(appearance, "applySkin", function()
		applySkin(State.skinUsername)
	end)
	addButton(appearance, "resetSkin", resetSkin)
	addButton(appearance, "randomSkin", randomSkin)
	addInput(appearance, "displayName", "displayName")
	addButton(appearance, "applyName", function()
		applyNameSpoof(State.displayName)
	end)
	addButton(appearance, "resetName", function()
		resetName()
		task.defer(rebuildUi)
	end)

	local visual = window:Tab({ Title = translate("visualSettings") })

	addToggle(visual, "enableAntiLag", "antiLagActive", applyAntiLag)
	addToggle(visual, "enableLighting", "lightingActive", applyLighting)
	addSlider(visual, "stretch", "stretchValue", 0, 100, 1, applyStretch)
	addButton(visual, "resetStretch", function()
		State.stretchValue = 0
		applyStretch()
		task.defer(rebuildUi)
	end)

	local themeLabels, currentTheme = {}, THEMES[1][2]

	for _, theme in ipairs(THEMES) do
		table.insert(themeLabels, theme[2])
		currentTheme = theme[1] == State.theme and theme[2] or currentTheme
	end

	addDropdown(visual, "theme", themeLabels, currentTheme, selectTheme)
	addSlider(visual, "camMult", "camMult", 1, 10, 0.1, applySensitivity)
	addSlider(visual, "reactionTime", "reactionTime", 0.05, 1, 0.05)

	local effects = window:Tab({ Title = translate("effects") })
	local currentEffect = EFFECT_LABELS[1]

	for _, label in ipairs(EFFECT_LABELS) do
		currentEffect = label:match("^%a+"):lower() == State.killEffect and label or currentEffect
	end

	addDropdown(effects, "killEffects", EFFECT_LABELS, currentEffect, function(value)
		State.killEffect = value:match("^%a+"):lower()
	end)
	addButton(effects, "testEffect", function()
		playKillEffect(State.killEffect, getCharacter())
	end)

	local settingsTab = window:Tab({ Title = translate("settings") })

	addDropdown(settingsTab, "selectLang", { LANGUAGES.ar, LANGUAGES.en }, LANGUAGES[State.language], function(value)
		local language = value:find("English") and "en" or "ar"

		-- WindUI fires the callback on creation, ignore it or the ui rebuilds forever.
		if language == State.language then
			return
		end

		State.language = language
		notify(translate("welcome"), translate("langChanged"))
		task.defer(rebuildUi)
	end)
	addToggle(settingsTab, "pinButton", "pinButton", applyPin)
	addToggle(settingsTab, "recordMode", "recordMode", applyRecordMode)
	addButton(settingsTab, "setKeybind", function()
		notify("Keybind", translate("pressKey"))

		local connection
		connection = userInputService.InputBegan:Connect(function(input, processed)
			if processed or input.KeyCode == Enum.KeyCode.Unknown then
				return
			end

			connection:Disconnect()
			setUIKeybind(input.KeyCode)
		end)
	end)
	settingsTab:Toggle({ Title = translate("stats"), Value = statsGui ~= nil, Callback = setStats })
	addButton(settingsTab, "resetPos", function()
		local root = getRoot()
		resetAssistPos()

		if root then
			root.AssemblyLinearVelocity = Vector3.zero
		end
	end)
	addButton(settingsTab, "closeScript", function()
		PhantomX.detach()
		notify("PhantomX", translate("scriptClosed"))
	end)

	local info = window:Tab({ Title = translate("info") })

	info:Paragraph({ Title = translate("developers"), Desc = "v" .. CONFIG.version })
	addButton(info, "joinDiscord", copyLink)
	addButton(info, "copyLink", copyLink)
	addButton(info, "contactDev", contactDev)
	info:Paragraph({ Title = "Version", Desc = CONFIG.version })

	local config = window:Tab({ Title = translate("savedSettings") })

	addButton(config, "save", saveConfig)
	addButton(config, "load", loadConfig)
	config:Paragraph({ Title = "Config Manager", Desc = translate("configManager") })

	return window
end

---Fetch and load the WindUI library.
loadWindUi = function()
	-- Download once, the source is reused for every rebuild.
	if not windUiSource then
		local response = httpRequest({ Url = CONFIG.windUi, Method = "GET" })
		local body = response and response.Body
		windUiSource = type(body) == "string" and #body > 0 and body or nil
	end

	if not windUiSource then
		warn("Failed to download WindUI.")
		return nil
	end

	cleanWindUiGuis()

	local success, library = pcall(function()
		return loadstring(windUiSource)()
	end)

	if not success then
		warn(library)
	end

	windUi = success and library or nil
	return windUi
end

---Destroy and rebuild the window so it matches the state and language.
rebuildUi = function()
	-- Never run two rebuilds at once.
	if rebuilding then
		return
	end

	rebuilding = true

	if window then
		pcall(window.Destroy, window)
		window = nil
	end

	-- A fresh library every time, WindUI can destroy its own gui when a window closes.
	if not loadWindUi() then
		rebuilding = false
		return notify("UI", translate("failedUI"))
	end

	local success, errorMessage = pcall(buildUi)

	if not success then
		warn(errorMessage)
		notify("UI", tostring(errorMessage):sub(1, 90))
	end

	applyTheme()
	rebuilding = false
end

-- Lifecycle.

---Toggle the window when the saved key is pressed.
local function onKeybindInput(input, processed)
	if not processed and State.uiKeybind ~= "" and input.KeyCode.Name == State.uiKeybind then
		toggleWindow()
	end
end

---Re-apply the appearance features after respawning.
local function onCharacterAdded()
	task.wait(0.6)
	table.clear(hidden)
	applyHeadless()
	applyNameSpoof()
	task.spawn(applyKorblox)
end

---Stop every loop, restore the game and remove our guis.
function PhantomX.detach()
	sessionToken, State.licensed = nil, false

	for name, connection in pairs(loops) do
		connection:Disconnect()
		loops[name] = nil
	end

	for _, connection in ipairs(connections) do
		connection:Disconnect()
	end

	table.clear(connections)

	if deathConnection then
		deathConnection:Disconnect()
		deathConnection = nil
	end

	-- Reset every applied feature.
	for _, flag in ipairs(RESET_FLAGS) do
		State[flag] = false
	end

	State.stretchValue, State.displayName, State.camBoost, State.camMult, State.pinButton = 0, "", 1, 1, false

	applySensitivity()
	applyAntiLag()
	applyLighting()
	applyStretch()
	applyHeadless()
	applyNameSpoof()
	applyPin()
	setStats(false)
	resetSpeed()
	destroyToggleButton()
	task.spawn(applyKorblox)

	if window then
		pcall(window.Destroy, window)
		window = nil
	end

	windUi = nil
	cleanWindUiGuis()
end

---Load the interface, start the loops and activate the license if there is a key.
function PhantomX.init()
	notify(translate("welcome"), translate("version") .. CONFIG.version, 4)
	task.wait(0.3)
	notify(translate("reminder"), translate("pray"), 3)

	local key = License.getKey()

	if key then
		task.spawn(License.activate, key)
	end

	rebuildUi()
	createToggleButton()

	track(userInputService.InputBegan:Connect(onKeybindInput))
	track(localPlayer.CharacterAdded:Connect(onCharacterAdded))
	setLoop("render", true, runService.RenderStepped, onRenderStep)
	setLoop("heartbeat", true, runService.Heartbeat, speedStep)
	notify("PhantomX", translate("ready"), 3)
end

---This is called when initialization errors.
---@param error string
local function onInitializeError(error)
	warn("Failed to initialize.")
	warn(error)
	warn(debug.traceback())
	PhantomX.detach()
end

-- Detach the previous instance before starting a new one.
if shared.PhantomX then
	shared.PhantomX.detach()
end

shared.PhantomX = PhantomX

-- Safely initialize the script and handle errors.
xpcall(PhantomX.init, onInitializeError)

return PhantomX
